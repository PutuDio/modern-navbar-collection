const nav=document.querySelector('.island'),head=document.querySelector('.head'),menu=document.querySelector('.menu'),search=document.querySelector('.search'),results=document.querySelector('.results');search.style.display='none';function leaveTimer;head.addEventListener('click',e=>open(e.pointerType==='mouse'?true:!nav.classList.contains('open')));nav.addEventListener('pointerenter',e=>{clearTimeout(leaveTimer);if(e.pointerType==='mouse')open(true)});nav.addEventListener('pointerleave',()=>{leaveTimer=setTimeout(()=>{if(search.style.display==='none'&&!nav.matches(':hover'))open(false)},650)});function select(label){document.querySelector('#active').textContent=label;document.querySelectorAll('[data-page]').forEach(b=>b.classList.toggle('active',b.dataset.page===label));closeSearch();open(false)}document.querySelectorAll('[data-page]').forEach(b=>b.addEventListener('click',()=>select(b.dataset.page)));function
